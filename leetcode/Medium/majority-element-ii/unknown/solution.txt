class Solution {
    public List<Integer> majorityElement(int[] nums) {
        Integer el1=null; 
        Integer el2=null;
        int cnt1=0;
        int cnt2=0;
        for(int num : nums){
            if(el1 != null &&  num==el1){
                cnt1++;

            }
            else if(el2 != null && num== el2){
                cnt2++;

            }
            else if(cnt1==0){
                el1=num;
                cnt1=1;

            }
            else if(cnt2==0){
                el2=num;
                cnt2=1;

            }
            else {
                cnt1--;
                cnt2--;

            }
        }
        cnt1=0;
        cnt2=0;

        for(int num :nums){
            if(el1 !=  null && num==el1)cnt1++;
            if(el2 != null && num== el2)cnt2++;
        }
            ArrayList <Integer>res=new ArrayList<>();
            int n=nums.length;


            if(cnt1> n/3)res.add(el1);
            if(cnt2>n/3)res.add(el2);

            return res;
        
    
        
    }
}
